   if(InpSets==QQX_SET_IC_RAW_HIGH) return (strategyNumber==1 || strategyNumber==8 || strategyNumber==10 || strategyNumber==12);

bool IsManagedPositionSelected(const int index)
{
   if(PositionGetString(POSITION_SYMBOL)!=_Symbol) return false;
   if(PositionGetInteger(POSITION_MAGIC)!=InpMagicNumber) return false;
   return StringFind(PositionGetString(POSITION_COMMENT),RecoveredTradeComment(index))==0;
}

int VolumeDigits(const double step)
{
   if(step<=0.0) return 2;
   int digits=0;
   double value=step;
   while(digits<8 && MathAbs(value-MathRound(value))>0.00000001)
   {
      value*=10.0;
      digits++;
   }
   return digits;
}

double NormalizeRecoveredVolume(double lots)
{
   double minimum=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MIN);
   double maximum=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MAX);
   double step=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_STEP);
   if(step<=0.0) step=0.01;
   lots=MathMax(minimum,MathMin(maximum,lots));
   lots=MathFloor((lots+0.000000001)/step)*step;
   return NormalizeDouble(MathMax(minimum,lots),VolumeDigits(step));
}

double RecoveredRequestedVolume()
{
   if(InpLotsCalc==QQX_LOTS_FIXED) return InpLotsFixed;
   if(InpLotsCalc==QQX_LOTS_FIXED_BALANCE)
   {
      if(InpLotsFixedBalance<=0.0) return 0.0;
      return AccountInfoDouble(ACCOUNT_BALANCE)/InpLotsFixedBalance*0.01;
   }
   // Curva ricavata con probe differenziali sul nativo a saldo 10k/100k.
   // Medium (3), usato dai preset originali, e verificato anche lungo tutta
   // la crescita del saldo: NormalizeDouble(balance/60000,2).
   double divisor=200000.0;
   if(InpAutoLotsValue==QQX_RISK_LOW) divisor=150000.0;
   else if(InpAutoLotsValue==QQX_RISK_LOW_MEDIUM) divisor=100000.0;
   else if(InpAutoLotsValue==QQX_RISK_MEDIUM) divisor=60000.0;
   else if(InpAutoLotsValue==QQX_RISK_MEDIUM_HIGH) divisor=60000.0;
   else if(InpAutoLotsValue==QQX_RISK_HIGH) divisor=55250.0;
   else if(InpAutoLotsValue==QQX_RISK_VERY_HIGH) divisor=40000.0;
   return NormalizeDouble(AccountInfoDouble(ACCOUNT_BALANCE)/divisor,2);
}

bool TradeResultAccepted()
{
   uint code=g_trade.ResultRetcode();
   return code==TRADE_RETCODE_DONE || code==TRADE_RETCODE_DONE_PARTIAL || code==TRADE_RETCODE_PLACED;
}

bool ReadRecoveredCycle(const int index,RecoveredCycle &cycle)
{
   cycle.count=0;
   cycle.direction=0;
   cycle.firstOpened=0;
   cycle.totalVolume=0.0;
   cycle.breakEven=0.0;
   cycle.lastReference=0.0;
   cycle.floatingProfit=0.0;
   double weighted=0.0;
   for(int i=PositionsTotal()-1;i>=0;i--)
   {
      ulong ticket=PositionGetTicket(i);
      if(ticket==0 || !PositionSelectByTicket(ticket) || !IsManagedPositionSelected(index)) continue;
      int direction=(PositionGetInteger(POSITION_TYPE)==POSITION_TYPE_BUY ? 1 : -1);
      if(cycle.direction!=0 && cycle.direction!=direction)
      {
         cycle.direction=2;
         cycle.count++;
         return true;
      }
      cycle.direction=direction;
      double volume=PositionGetDouble(POSITION_VOLUME);
      double price=PositionGetDouble(POSITION_PRICE_OPEN);
      datetime opened=(datetime)PositionGetInteger(POSITION_TIME);
      cycle.count++;
      if(cycle.firstOpened==0 || opened<cycle.firstOpened) cycle.firstOpened=opened;
      cycle.totalVolume+=volume;
      cycle.floatingProfit+=PositionGetDouble(POSITION_PROFIT)+PositionGetDouble(POSITION_SWAP);
      weighted+=price*volume;
      if(cycle.count==1) cycle.lastReference=price;
      else if(direction>0 && price<cycle.lastReference) cycle.lastReference=price;
      else if(direction<0 && price>cycle.lastReference) cycle.lastReference=price;
   }
   if(cycle.count<=0 || cycle.totalVolume<=0.0) return false;
   cycle.breakEven=weighted/cycle.totalVolume;
   return true;
}

bool CloseRecoveredCycle(const int index)
{
   if(!RecoveredMutationAuthorized()) return false;
   bool ok=true;
   ulong tickets[];
   long opened[];
   int count=0;
   for(int i=0;i<PositionsTotal();i++)
   {
      ulong ticket=PositionGetTicket(i);
      if(ticket==0 || !PositionSelectByTicket(ticket) || !IsManagedPositionSelected(index)) continue;
      ArrayResize(tickets,count+1);
      ArrayResize(opened,count+1);
      tickets[count]=ticket;
      opened[count]=(long)PositionGetInteger(POSITION_TIME_MSC);
      count++;
   }
   for(int i=1;i<count;i++)
   {
      ulong ticketKey=tickets[i];
      long timeKey=opened[i];
      int j=i-1;
      while(j>=0 && (opened[j]>timeKey || (opened[j]==timeKey && tickets[j]>ticketKey)))
      {
         opened[j+1]=opened[j];
         tickets[j+1]=tickets[j];
         j--;
      }
      opened[j+1]=timeKey;
      tickets[j+1]=ticketKey;
   }
   for(int i=0;i<count;i++)
   {
      ResetLastError();
      if(!g_trade.PositionClose(tickets[i]) || !TradeResultAccepted())
      {
         ok=false;
         Print("RECOVERED ",RecoveredStrategyTag(index)," close failed ticket=",tickets[i],
               " retcode=",g_trade.ResultRetcode()," error=",GetLastError());
      }
   }
   return ok;
}

bool CloseAllRecoveredCycles()
{
   bool ok=true;
   for(int index=0;index<RECOVERED_STRATEGY_COUNT;index++)
      if(!CloseRecoveredCycle(index)) ok=false;
   return ok;
}

bool DirectionAllowed(const int index,const int direction)
{
   if(g_profiles[index].nativeDirection>0 && direction<0) return false;
   if(g_profiles[index].nativeDirection<0 && direction>0) return false;
   if(InpTradingDirectionType==QQX_BUYS_ONLY) return direction>0;
   if(InpTradingDirectionType==QQX_SELLS_ONLY) return direction<0;
   return direction==1 || direction==-1;
}

bool RecoveredSessionHourAllowed(const int index,const int hour)
{
   return hour>=g_profiles[index].sessionStart && hour<g_profiles[index].sessionEnd;
}

bool RecoveredWeekdayAllowed(const int weekday)
{
   if(weekday==0) return InpTradeOnSunday;
   if(weekday==1) return InpTradeOnMonday;
   if(weekday==2) return InpTradeOnTuesday;
   if(weekday==3) return InpTradeOnWednesday;
   if(weekday==4) return InpTradeOnThursday;
   if(weekday==5) return InpTradeOnFriday;
   if(weekday==6) return InpTradeOnSaturday;
   return false;
}

bool IsLikelyNfpFriday(const MqlDateTime &now)
{
   if(now.day_of_week!=5) return false;
   return (now.day<=7);
}

int RecoveredOrdersMaxForStrategy(const int index)
{
   int strategyNumber=g_profiles[index].strategyNumber;
   if(strategyNumber==9) return MathMin(InpOrdersMax,8);
   if(strategyNumber==3) return MathMin(InpOrdersMax,10);
   if(strategyNumber==6 || strategyNumber==10) return MathMin(InpOrdersMax,15);
   return InpOrdersMax;
}

bool DrawdownPctExceeded()
{
   if(InpDDMode==QQX_DD_OFF || InpDDValue<=0.0) return false;
   if(InpDDMode!=QQX_DD_PCT_CONTINUE && InpDDMode!=QQX_DD_PCT_REMOVE && InpDDMode!=QQX_DD_PCT_ALERT)
      return false;
   double equity=AccountInfoDouble(ACCOUNT_EQUITY);
   if(equity>g_equityPeak) g_equityPeak=equity;
   if(g_equityPeak<=0.0) return false;
   double ddPct=100.0*(g_equityPeak-equity)/g_equityPeak;
   return ddPct>=InpDDValue;
}

bool DrawdownMoneyExceeded()
{
   if(InpDDMode==QQX_DD_OFF || InpDDValue<=0.0) return false;
   if(InpDDMode!=QQX_DD_MONEY_CONTINUE && InpDDMode!=QQX_DD_MONEY_REMOVE && InpDDMode!=QQX_DD_MONEY_ALERT)
      return false;
   double equity=AccountInfoDouble(ACCOUNT_EQUITY);
   if(equity>g_equityPeak) g_equityPeak=equity;
   return (g_equityPeak-equity)>=InpDDValue;
}

void HandleDrawdownProtection()
{
   if(g_drawdownLock) return;
   if(!DrawdownPctExceeded() && !DrawdownMoneyExceeded()) return;
   CloseAllRecoveredCycles();
   if(InpDDMode==QQX_DD_PCT_ALERT || InpDDMode==QQX_DD_MONEY_ALERT)
      Alert("QQX drawdown limit reached: ",DoubleToString(InpDDValue,2));
   if(InpDDMode==QQX_DD_PCT_REMOVE || InpDDMode==QQX_DD_MONEY_REMOVE)
      g_drawdownLock=true;
}

bool LocalEntryGates(const int index)
{
   if(!RecoveredAutomationAuthorized() || !RecoveredStrategyEnabled(index) || !IsHedgingAccount()) return false;
   MqlTick tick;
   if(!SymbolInfoTick(_Symbol,tick)) return false;
   if((tick.ask-tick.bid)/_Point>InpSpread) return false;
   MqlDateTime now;
   if(!TimeToStruct(TimeCurrent(),now)) return false;
   if(!RecoveredWeekdayAllowed(now.day_of_week)) return false;
   if(InpUseNfpFridayFilter && IsLikelyNfpFriday(now)) return false;
   if(!RecoveredSessionHourAllowed(index,now.hour)) return false;
   if(now.day_of_week==5 && InpTradingFridayNight && now.hour>=InpTradingFridayNightHour) return false;
   if(InpUseYearEndPause && ((now.mon==12 && now.day>=15) || (now.mon==1 && now.day<=15))) return false;
   return true;
}

bool OpenRecoveredDirection(const int index,const int direction)
{
   if(!RecoveredAutomationAuthorized() || !DirectionAllowed(index,direction)) return false;
   MqlTick tick;
   if(!SymbolInfoTick(_Symbol,tick)) return false;
   double volume=NormalizeRecoveredVolume(RecoveredRequestedVolume());
   if(volume<=0.0) return false;
   ENUM_ORDER_TYPE type=(direction>0 ? ORDER_TYPE_BUY : ORDER_TYPE_SELL);
   double price=(direction>0 ? tick.ask : tick.bid);
   ResetLastError();
   bool opened=g_trade.PositionOpen(_Symbol,type,volume,price,0.0,0.0,RecoveredTradeComment(index));
   opened=opened && TradeResultAccepted();
   if(!opened)
      Print("RECOVERED ",RecoveredStrategyTag(index)," open failed retcode=",g_trade.ResultRetcode(),
            " ",g_trade.ResultRetcodeDescription()," error=",GetLastError());
   return opened;
}

double RecoveredGridBasePoints(const int index)
{
   double scaled=g_profiles[index].gridBaseRaw*MathPow(10.0,2-_Digits);
   return MathMax(1.0,MathFloor(scaled));
}

double RecoveredTakeProfitDateMultiplier(const datetime moment=0)
{
   // Moltiplicatori per-date rimossi: erano date hardcoded (5x/15x TP) che
   // gonfiavano il backtest storico rispetto al live.
   return 1.0;
}

void ManageRecoveredCycle(const int index,const RecoveredCycle &cycle)
{
   if(!RecoveredAutomationAuthorized()) return;
   if(cycle.direction!=1 && cycle.direction!=-1) return;
   MqlTick tick;
   if(!SymbolInfoTick(_Symbol,tick)) return;
   RecoveredProfile profile=g_profiles[index];
   int level=MathMax(0,cycle.count-1);
   double tpDistance=(profile.tpBase*RecoveredTakeProfitDateMultiplier(cycle.firstOpened)+profile.tpIncrement*level)*_Point;
   double tp=NormalizeDouble(cycle.direction>0 ? cycle.breakEven+tpDistance : cycle.breakEven-tpDistance,_Digits);
   double sl=0.0;
   if(profile.slDistance>0)
      sl=NormalizeDouble(cycle.direction>0 ? cycle.breakEven-profile.slDistance*_Point
                                           : cycle.breakEven+profile.slDistance*_Point,_Digits);
   if(g_trailingDirection[index]!=cycle.direction)
   {
      g_trailingDirection[index]=cycle.direction;
      g_trailingStop[index]=(profile.trailingDistance>0 && sl>0.0 ? sl : 0.0);
   }
   if(profile.trailingDistance>0 && sl>0.0 && g_trailingStop[index]==0.0) g_trailingStop[index]=sl;
   if(profile.trailingDistance>0 && g_trailingStop[index]>0.0)
   {
      if(cycle.direction>0)
      {
         double candidate=NormalizeDouble(tick.bid-profile.trailingDistance*_Point,_Digits);
         if(candidate>g_trailingStop[index] && candidate>cycle.breakEven+profile.trailingGuard*_Point)
            g_trailingStop[index]=candidate;
      }
      else
      {
         double candidate=NormalizeDouble(tick.ask+profile.trailingDistance*_Point,_Digits);
         if(g_trailingStop[index]>candidate && candidate<cycle.breakEven-profile.trailingGuard*_Point)
            g_trailingStop[index]=candidate;
      }
   }
   bool takeProfit=(cycle.direction>0 ? tick.bid>tp : tp>tick.ask);
   bool stopLoss=(sl>0.0 && (cycle.direction>0 ? sl>tick.ask : tick.bid>sl));
   bool trailing=(g_trailingStop[index]>0.0 &&
                  (cycle.direction>0 ? g_trailingStop[index]>tick.ask : tick.bid>g_trailingStop[index]));
   if(takeProfit || stopLoss || trailing)
   {
      CloseRecoveredCycle(index);
      return;
   }
   if(cycle.count>=RecoveredOrdersMaxForStrategy(index)) return;
   double gridDistance=(RecoveredGridBasePoints(index)+profile.gridIncrement*level)*_Point;
   double grid=NormalizeDouble(cycle.direction>0 ? cycle.lastReference-gridDistance
                                                 : cycle.lastReference+gridDistance,_Digits);
   bool add=(cycle.direction>0 ? grid>tick.ask : tick.bid>grid);
   if(add) OpenRecoveredDirection(index,cycle.direction);
}

bool ReadRecoveredDemarker(const int handle,double &value)
{
   if(handle==INVALID_HANDLE) return false;
   double values[];
   ArrayResize(values,2);
   if(CopyBuffer(handle,0,0,2,values)!=2) return false;
   value=values[0];
   return MathIsValidNumber(value);
}

int ThresholdDirection(const double value,const double upper,const double lower)
{
   if(value>upper) return 1;
   if(value<lower) return -1;
   return 0;
}

bool IsNewM1Bar()
{
   datetime current=iTime(_Symbol,PERIOD_M1,0);
   if(current<=0 || current==g_lastM1Bar) return false;
   g_lastM1Bar=current;
   return true;
}

bool IsNewEntryBar(const int index)
{
   datetime current=iTime(_Symbol,g_profiles[index].firstTf,0);
   if(current<=0 || current==g_lastEntryBar[index]) return false;
   g_lastEntryBar[index]=current;
   return true;
}

void TryRecoveredEntry(const int index)
{
   if(!LocalEntryGates(index)) return;
   RecoveredCycle cycle;
   if(ReadRecoveredCycle(index,cycle)) return;
   double firstValue=0.0,secondValue=0.0;
   if(!ReadRecoveredDemarker(g_firstHandle[index],firstValue)) return;
   if(!ReadRecoveredDemarker(g_secondHandle[index],secondValue)) return;
   int firstSignal=ThresholdDirection(firstValue,g_profiles[index].firstUpper,g_profiles[index].firstLower);
   int secondSignal=ThresholdDirection(secondValue,g_profiles[index].secondUpper,g_profiles[index].secondLower);
   if(firstSignal==0 || firstSignal!=secondSignal) return;
   if(!DirectionAllowed(index,firstSignal)) return;
   OpenRecoveredDirection(index,firstSignal);
}

int OnInit()
{
   bool nonVisualTest=(MQLInfoInteger(MQL_TESTER) && !MQLInfoInteger(MQL_VISUAL_MODE));
   if(MQLInfoInteger(MQL_TESTER))
      g_testerTradingOverride=true;
   g_panelPaused=(InpPause==QQX_START_PAUSED && !nonVisualTest);
   g_equityPeak=AccountInfoDouble(ACCOUNT_EQUITY);
   SetRecoveredProfiles();
   for(int index=0;index<RECOVERED_STRATEGY_COUNT;index++)
   {
      g_firstHandle[index]=iDeMarker(_Symbol,g_profiles[index].firstTf,g_profiles[index].firstPeriod);
      g_secondHandle[index]=iDeMarker(_Symbol,g_profiles[index].secondTf,g_profiles[index].secondPeriod);
      if(g_firstHandle[index]==INVALID_HANDLE || g_secondHandle[index]==INVALID_HANDLE)
      {
         Print("RECOVERED ",RecoveredStrategyTag(index),": creazione handle iDeMarker fallita, error=",GetLastError());
         return INIT_FAILED;
      }
      g_lastEntryBar[index]=iTime(_Symbol,g_profiles[index].firstTf,0);
      g_trailingStop[index]=0.0;
      g_trailingDirection[index]=0;
   }
   if(InpEnableRecoveredTrading && !IsHedgingAccount())
   {
      Print("RECOVERED portfolio richiede un conto hedging");
      return INIT_PARAMETERS_INCORRECT;
   }
   g_trade.SetExpertMagicNumber((ulong)MathMax((long)0,InpMagicNumber));
   g_trade.SetDeviationInPoints((ulong)MathMax(0,InpSlippage));
   g_trade.SetTypeFillingBySymbol(_Symbol);
   g_lastM1Bar=iTime(_Symbol,PERIOD_M1,0);
   if(RecoveredPanelAllowed())
   {
      EventSetTimer(1);
      UpdateRecoveredPanel();
   }
   Print("BinScope Quantum Queen portfolio S01-S12; trading=",InpEnableRecoveredTrading,
         "; opaque gate acknowledged=",InpAcknowledgeOpaqueGate);
   return INIT_SUCCEEDED;
}

void OnChartEvent(const int id,const long &lparam,const double &dparam,const string &sparam)
{
   if(id!=CHARTEVENT_OBJECT_CLICK) return;
   string prefix=RecoveredPanelPrefix();
   if(StringFind(sparam,prefix)!=0) return;
   ObjectSetInteger(0,sparam,OBJPROP_STATE,false);
   if(sparam==prefix+"PAUSE")
   {
      g_panelPaused=!g_panelPaused;
      UpdateRecoveredPanel();
      return;
   }
   if(sparam==prefix+"INFO")
   {
      MessageBox("Quantum Queen X MT5 - semantic portfolio reconstruction\n"
                 "Twelve native modules recovered; S11 remains dormant.\n"
                 "TP, BreakEven and Grid lines are virtual.",InpComment,MB_OK);
      return;
   }
   if(sparam==prefix+"CLOSE_ALL")
   {
      int answer=MessageBox("Are you sure you want to CLOSE ALL TRADES managed by the EA?\n"
                            "Only positions on this symbol and Magic Number are affected.",
                            InpComment,MB_YESNO);
      if(answer==IDYES) CloseAllRecoveredCycles();
      UpdateRecoveredPanel();
   }
}

void OnTick()
{
   if(InpDDMode!=QQX_DD_OFF) HandleDrawdownProtection();
   if(g_drawdownLock) return;
   bool newM1=IsNewM1Bar();
   bool newEntryBar[RECOVERED_STRATEGY_COUNT];
   bool hadCycle[RECOVERED_STRATEGY_COUNT];
   RecoveredCycle cycles[RECOVERED_STRATEGY_COUNT];
   for(int index=0;index<RECOVERED_STRATEGY_COUNT;index++)
   {
      // Il nativo consuma la transizione della timeframe di ingresso anche
      // mentre il basket e aperto: dopo una chiusura non puo quindi rientrare
      // a meta della stessa barra.
      newEntryBar[index]=IsNewEntryBar(index);
      ZeroMemory(cycles[index]);
      hadCycle[index]=ReadRecoveredCycle(index,cycles[index]);
   }
   // Il nativo valuta prima le nuove entrate dei moduli liberi e solo dopo
   // la gestione M1 dei basket gia aperti. Questo determina l'ordine ticket
   // quando, nello stesso tick, S10 entra e S01 aggiunge una griglia.
   for(int index=0;index<RECOVERED_STRATEGY_COUNT;index++)
   {
      if(hadCycle[index]) continue;
      g_trailingStop[index]=0.0;
      g_trailingDirection[index]=0;
      if(newEntryBar[index]) TryRecoveredEntry(index);
   }
   if(!newM1) return;
   for(int index=0;index<RECOVERED_STRATEGY_COUNT;index++)
      if(hadCycle[index]) ManageRecoveredCycle(index,cycles[index]);
}
